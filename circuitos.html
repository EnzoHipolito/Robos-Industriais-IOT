01 Exercício

Exercício 1

Monte um circuito capaz de controlar automaticamente o LED utilizando as informações dos dois sensores.
O sistema deverá seguir esta lógica:
• Pessoa presente + ambiente escuro → LED ligado
• Pessoa presente + ambiente claro → LED desligado
• Sem pessoa → LED desligado

Circuito no TinkerCAD

Código Arduino

const int pinPIR = 2;
const int pinLDR = A0;
const int pinLED = 8;

int limiarLuminosidade = 400; // Ajuste conforme a simulação do Tinkercad

void setup() {
  pinMode(pinPIR, INPUT);
  pinMode(pinLED, OUTPUT);
  pinMode(pinLDR, INPUT);
  Serial.begin(9600);
}

void loop() {
  int presenca = digitalRead(pinPIR);
  int leituraLuminosidade = analogRead(pinLDR);

  // Lógica: Pessoa presente (HIGH) E Ambiente Escuro (leitura < limiar)
  if (presenca == HIGH && leituraLuminosidade < limiarLuminosidade) {
    digitalWrite(pinLED, HIGH);
  } else {
    digitalWrite(pinLED, LOW);
  }
  delay(100);
}
02 Exercício

Exercício 2

Desenvolva um sistema de estacionamento utilizando o sensor ultrassônico para medir a distância e o potenciômetro para definir o limite de segurança.
Regras de funcionamento: O potenciômetro deverá determinar a distância limite.
Exemplo: • Objeto distante → LED desligado
• Objeto se aproximando → LED piscando
• Objeto dentro da distância definida → LED ligado
• O valor configurado pelo potenciômetro deverá influenciar o comportamento do sistema.

Código Arduino

const int pinoSensor = 7;
const int pinoLED = 4;
const int pinoPotenciometro = A0;

// Distância mínima considerada para o alerta máximo
const int distanciaCritica = 10;

// Variáveis
long duracao;
float distancia;

void setup() {
  pinMode(pinoSensor, OUTPUT);
  pinMode(pinoLED, OUTPUT);
  Serial.begin(9600);
}

void loop() {

  // ==========================================
  // LEITURA DO POTENCIÔMETRO
  // ==========================================

  int valorPot = analogRead(pinoPotenciometro);

  // Converte o potenciômetro para uma distância
  // de segurança entre 10 cm e 50 cm
  int distanciaLimite = map(valorPot, 0, 1023, 10, 50);

  // ==========================================
  // LEITURA DO SENSOR ULTRASSÔNICO
  // ==========================================

  // Envia o pulso
  pinMode(pinoSensor, OUTPUT);
  digitalWrite(pinoSensor, LOW);
  delayMicroseconds(2);
  digitalWrite(pinoSensor, HIGH);
  delayMicroseconds(5);
  digitalWrite(pinoSensor, LOW);

  // Muda o pino para entrada para receber o retorno
  pinMode(pinoSensor, INPUT);
  duracao = pulseIn(pinoSensor, HIGH);

  // Converte o tempo em distância
  distancia = duracao / 29.0 / 2.0;

  // ==========================================
  // EXIBE OS VALORES NO MONITOR SERIAL
  // ==========================================

  Serial.print("Distancia: ");
  Serial.print(distancia);
  Serial.print(" cm | Limite: ");
  Serial.print(distanciaLimite);
  Serial.println(" cm");

  // ==========================================
  // SISTEMA DE ALERTA
  // ==========================================

  // OBJETO MUITO LONGE
  if (distancia > distanciaLimite) {
    digitalWrite(pinoLED, LOW);
    delay(50);
  }

  // OBJETO SE APROXIMANDO
  else if (distancia > distanciaCritica) {

    // Quanto mais perto, menor o intervalo entre as piscadas.
    int intervalo = map(
      distancia,
      distanciaLimite,
      distanciaCritica,
      700,
      100
    );
    digitalWrite(pinoLED, HIGH);
    delay(intervalo);
    digitalWrite(pinoLED, LOW);
    delay(intervalo);
  }

  // OBJETO MUITO PRÓXIMO
  else {
    digitalWrite(pinoLED, HIGH);
    delay(50);
  }
}

03 Exercício

Exercício 3

Desenvolva um sistema capaz de monitorar a temperatura e a umidade da estufa.
Os LEDs deverão funcionar como indicadores visuais das condições do ambiente.
Regras de funcionamento: O aluno deverá estabelecer valores de referência para temperatura e umidade.
Exemplo: Condição adequada
Temperatura e umidade dentro dos valores definidos:
→ LED verde ligado.
Temperatura elevada
Temperatura acima do limite:
→ LED vermelho ligado.
Umidade inadequada
Umidade abaixo do limite:
→ LED amarelo ligado.

Circuito no TinkerCAD

Código Arduino

// LEDs
const int ledVerde = 12;
const int ledAmarelo = 7;
const int ledVermelho = 4;

// Sensores
const int sensorTemperatura = A0;
const int sensorUmidade = A1;

// Valores de referência
const float limiteTemperatura = 30.0; // °C
const int limiteUmidade = 40;         // %

void setup() {
  pinMode(ledVerde, OUTPUT);
  pinMode(ledAmarelo, OUTPUT);
  pinMode(ledVermelho, OUTPUT);
  Serial.begin(9600);
}

void loop() {

  // ==========================================
  // LEITURA DA TEMPERATURA - TMP36
  // ==========================================

  int valorTemperatura = analogRead(sensorTemperatura);
  float tensao = valorTemperatura * (5.0 / 1023.0);
  float temperatura = (tensao - 0.5) * 100.0;

  // ==========================================
  // LEITURA DA UMIDADE
  // ==========================================

  int valorUmidade = analogRead(sensorUmidade);
  int umidade = map(valorUmidade, 0, 1023, 0, 100);
  umidade = constrain(umidade, 0, 100);

  // ==========================================
  // MONITOR SERIAL
  // ==========================================

  Serial.print("Temperatura: ");
  Serial.print(temperatura);
  Serial.print(" °C | Umidade: ");
  Serial.print(umidade);
  Serial.println("%");

  // ==========================================
  // VERIFICA AS CONDIÇÕES
  // ==========================================

  bool temperaturaAlta = temperatura > limiteTemperatura;
  bool umidadeBaixa = umidade < limiteUmidade;

  // ==========================================
  // DESLIGA TODOS OS LEDs
  // ==========================================

  digitalWrite(ledVerde, LOW);
  digitalWrite(ledAmarelo, LOW);
  digitalWrite(ledVermelho, LOW);

  // TUDO ADEQUADO
  if (!temperaturaAlta && !umidadeBaixa) {
    digitalWrite(ledVerde, HIGH);
  }

  // TEMPERATURA ALTA
  if (temperaturaAlta) {
    digitalWrite(ledVermelho, HIGH);
  }

  // UMIDADE BAIXA
  if (umidadeBaixa) {
    digitalWrite(ledAmarelo, HIGH);
  }

  delay(1000);
}
